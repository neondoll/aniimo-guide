    public function actionAll(): array
    {
        $candidates = [];
        $organizations = [];

        // $lastCompletedCommission = Commissions::find()->where(['status' => Commissions::STATUSES['CommissionIsCompleted'], 'system_status' => 1])->orderBy(['id' => SORT_DESC])->one();
        // $lastInSessionCommission = Commissions::find()->where(['status' => Commissions::STATUSES['CommissionIsInSession'], 'system_status' => 1])->orderBy(['id' => SORT_DESC])->with(['orgs'])->one();

        $inSessionCommissions = Commissions::find()->where(['status' => Commissions::STATUSES['CommissionIsInSession'], 'system_status' => 1])->with(['orgs'])->all();

        // $users = User::find()->where([
        //     'AND',
        //     ['id_org' => array_map(function (OrganizationCommission $org) {
        //         return $org->id_org;
        //     }, $lastInSessionCommission->orgs)],
        //     ['>=', 'created_at', "$lastCompletedCommission->date 00:00:00"],
        // ])->all();

        $conditions = [];

        foreach ($inSessionCommissions as $commission) {
            $conditions[] = [
                'AND',
                ['id_org' => array_map(function (OrganizationCommission $oc) { return $oc->id_org; }, $commission->orgs)],
                ['>=', 'created_at', $commission->date . ' 00:00:00'],
            ];
        }

        if (empty($conditions)) {
            return ['candidates' => [], 'organizations' => []];
        }

        $query = User::find();

        if (count($conditions) === 1) $query->where($conditions[0]);
        else $query->where(array_merge(['OR'], $conditions));

        $users = $query->all();

        foreach ($users as $user) {
            $organizationId = $user->id_org ?? 0;
            $candidates[$organizationId][] = $user;

            if (!array_key_exists($organizationId, $organizations)) {
                $organizations[$organizationId] = Organizations::findOne($organizationId) ?? ['name' => 'Не задана'];
            }
        }

        return ['candidates' => $candidates, 'organizations' => $organizations];
    }
